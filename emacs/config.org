#+title: Emacs Config
#+property: header-args :tangle init.el
Guix Home tangles this file at build time (see =home.scm=):
=init.el= and =early-init.el= land in =~/.config/emacs/=, =exwm.el=
lands in =~/.exwm=. All packages come from Guix; the few Guix lacks are
fetched with =use-package :vc=.

** mu4e
Driven from Emacs: mu4e runs =mbsync= on its own timer (no daemon),
sends with the built-in =smtpmail=, and reads the password from pass via
=auth-source-pass=. =mbsync='s config is generated in =home.scm=.

PLACEHOLDER: replace =YOU@riseup.net= / =YOUR NAME= here and
=%mail-address= in =home.scm=.
#+begin_src emacs-lisp
(defvar my/mail-address "YOU@riseup.net")
(defvar my/mail-host "mail.riseup.net")

(use-package mu4e
  :commands mu4e
  :custom
  (user-mail-address my/mail-address)
  (user-full-name "YOUR NAME")
  (mu4e-get-mail-command "mbsync -a")
  (mu4e-update-interval (* 10 60))
  ;; mbsync needs new file names when mu4e moves messages
  (mu4e-change-filenames-when-moving t)
  (mu4e-sent-folder "/riseup/Sent")
  (mu4e-drafts-folder "/riseup/Drafts")
  (mu4e-trash-folder "/riseup/Trash")
  (mu4e-refile-folder "/riseup/Archive")
  (mu4e-maildir-shortcuts
   '((:maildir "/riseup/INBOX" :key ?i)
     (:maildir "/riseup/Sent" :key ?s)
     (:maildir "/riseup/Archive" :key ?a)))
  (message-send-mail-function #'smtpmail-send-it)
  (message-kill-buffer-on-exit t)
  (smtpmail-smtp-server my/mail-host)
  (smtpmail-smtp-service 465)
  (smtpmail-stream-type 'ssl)
  (smtpmail-smtp-user my/mail-address)
  :config
  ;; Decrypt the password in Emacs first (minibuffer prompt) so gpg-agent
  ;; has it cached by the time mbsync's PassCmd runs `pass'.
  (add-hook 'mu4e-update-pre-hook
            (lambda () (auth-source-pass-get 'secret my/mail-host))))
#+end_src

(defun my/kill-buffer-and-frame-or-exit (&optional arg)
  "Kill this buffer and frame.  With only one frame, exit Emacs.

When there is more than one visible frame, run `delete-frame'.
Otherwise, exit Emacs with `save-buffers-kill-terminal' after
confirming this with user.

If a prefix ARG is supplied, ignore it in the multiple-frame
case.  Otherwise, bypass confirmation and pass the argument to

   ((or arg (y-or-n-p "Quit Emacs?"))    (save-buffers-kill-terminal arg))

(defadvice expand-abbrev (after my-expand-abbrev activate)
  ;; if there was an expansion
  (if ad-return-value
      ;; start idle timer to ensure insertion of abbrev activator
      ;; character (e.g. space) is finished
      (run-with-idle-timer
       0 nil
       (lambda ()
         ;; if there is the string "@@" in the
         ;; expansion then move cursor there and
         ;; delete the string
         (let ((cursor "@@"))
           (if (search-backward cursor last-abbrev-location t)
                                   (delete-char (length cursor))))))))

* exwm
Tangled to =exwm.el=, which Guix Home links to =~/.exwm=. The =exwm=
launcher from the Guix =emacs-exwm= package (the "EXWM" session in GDM)
loads it after =init.el=, so plain =emacs= started elsewhere never
tries to become a window manager.

#+begin_src emacs-lisp :tangle exwm.el
;; -*- lexical-binding: t -*-
(require 'exwm)

(setq exwm-workspace-number 4)
#+end_src

** buffer names
X windows show up as buffers; name them after the program, and after
the page/file title for the browser and mpv.
#+begin_src emacs-lisp :tangle exwm.el
(add-hook 'exwm-update-class-hook
          (lambda () (exwm-workspace-rename-buffer exwm-class-name)))
(add-hook 'exwm-update-title-hook
          (lambda ()
            (when (member exwm-class-name '("qutebrowser" "mpv"))
              (exwm-workspace-rename-buffer exwm-title))))
#+end_src

** global keys
Super comes from the kmonad home-row mod on =s=. =s-r= stays with crux,
so reset is on =s-R=.
#+begin_src emacs-lisp :tangle exwm.el
(defun my/exwm-run (command)
  "Start shell COMMAND without an output buffer."
  (interactive (list (read-shell-command "$ ")))
  (start-process-shell-command command nil command))

(setq exwm-input-global-keys
      `(([?\s-R] . exwm-reset)
        ([?\s-w] . exwm-workspace-switch)
        ([?\s-&] . my/exwm-run)
        ,@(mapcar (lambda (i)
                    `(,(kbd (format "s-%d" i))
                      . (lambda ()
                          (interactive)
                          (exwm-workspace-switch-create ,i))))
                  (number-sequence 0 9))))
#+end_src

** laptop keys
=wpctl= comes with PipeWire, =brightnessctl= has a udev rule in =system.scm=.
#+begin_src emacs-lisp :tangle exwm.el
(dolist (binding
         '(("<XF86AudioRaiseVolume>" . "wpctl set-volume -l 1.0 @DEFAULT_AUDIO_SINK@ 5%+")
           ("<XF86AudioLowerVolume>" . "wpctl set-volume @DEFAULT_AUDIO_SINK@ 5%-")
           ("<XF86AudioMute>" . "wpctl set-mute @DEFAULT_AUDIO_SINK@ toggle")
           ("<XF86MonBrightnessUp>" . "brightnessctl set 5%+")
           ("<XF86MonBrightnessDown>" . "brightnessctl set 5%-")))
  (let ((command (cdr binding)))
    (exwm-input-set-key (kbd (car binding))
                        (lambda () (interactive) (my/exwm-run command)))))
#+end_src
